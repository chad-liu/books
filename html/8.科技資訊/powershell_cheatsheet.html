函式定義・參數
function Greet {
  param(
    [Parameter(Mandatory)]
    [string]$Name,
    [string]$City = "Kaohsiung",
    [ValidateSet("Admin","User")]
    [string]$Role = "User",
    [switch]$IsActive
  )
  "$Name / $City / $Role"
}
Greet -Name "Chuck" -IsActive
回傳值
# 自動回傳（未捕捉的輸出）
function Get-Square($n) { $n * $n }

# 明確 return
function Get-Max($a,$b) {
  if ($a -gt $b) { return $a }
  return $b
}

# 回傳多值
$min, $max = Get-MinMax @(3,1,9)
Pipeline 支援
function Double-Value {
  param(
    [Parameter(ValueFromPipeline)]
    [int]$Number
  )
  begin   { # 執行一次 }
  process { $Number * 2 }
  end     { # 結束後 }
}
1..5 | Double-Value
錯誤處理

  Get-Content $Path -ErrorAction Stop
}

  Write-Error "找不到：$Path"
}
catch {
  Write-Error $_.Exception.Message
}
finally {
  Write-Host "完畢"
}
